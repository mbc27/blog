部署
# 全局安装 PM2
npm install -g pm2

# 进入后端 jar 文件所在目录
cd 你的jar文件目录

# 创建后端启动脚本
echo java -jar blog-api-0.0.1-SNAPSHOT.jar > start-backend.bat


pm2 start start-backend.bat --name blog-backend


# 进入前端项目目录
cd frontend/blog

# 启动前端服务
pm2 start "serve -s dist -l 8081" --name blog-frontend




# 生成开机启动脚本
pm2 startup

# 保存当前进程列表
pm2 save



=========================================
常用
pm2 status          # 查看所有进程状态
pm2 list            # 列出所有进程



pm2 logs                    # 查看所有进程日志
pm2 logs blog-backend       # 查看后端日志
pm2 logs blog-frontend      # 查看前端日志



pm2 restart blog-backend    # 重启后端
pm2 restart blog-frontend   # 重启前端
pm2 restart all             # 重启所有服务

pm2 stop blog-backend       # 停止后端
pm2 stop blog-frontend      # 停止前端
pm2 stop all                # 停止所有服务



pm2 delete blog-backend     # 删除后端进程
pm2 delete blog-frontend    # 删除前端进程
pm2 delete all              # 删除所有进程



pm2 monit                   # 实时监控进程状态
